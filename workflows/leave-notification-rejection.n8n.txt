// Return only reconciliation metadata, never a current/fixed recipient fallback
// or the rejected request's HR payload.
return [{ json: {
  ok: false, sent: false, alreadySent: false, error: 'Request rejected.',
  rejectionCode: String($json.rejectionCode ?? 'missing'),
  ...(typeof $json.reconciliationRequired === 'boolean' ? {
    reconciliationRequired: $json.reconciliationRequired,
    eventKey: $json.eventKey,
    ...($json.receiptEventKey !== undefined ? { receiptEventKey: $json.receiptEventKey } : {}),
    ...(typeof $json.actualRecipient === 'string' ? { actualRecipient: $json.actualRecipient } : {}),
    ...(typeof $json.actualCcRecipients === 'string' ? { actualCcRecipients: $json.actualCcRecipients } : {}),
    ...($json.providerReceipt?.accepted === true ? { providerReceipt: $json.providerReceipt } : {}),
  } : {}),
} }];
