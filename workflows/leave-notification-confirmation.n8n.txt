const sent = $input.all();
// Explicit failure stops BEFORE the confirm HTTP node. Unknown/lost responses
// leave the durable reservation processing; there is no timeout takeover.
if (sent.some(item => {
  const result = item.json ?? {};
  return Boolean(result.error) || result.success === false || result.ok === false || result.accepted === false
    || [result.statusCode, result.status].some(status => status !== undefined && status !== null
      && !(Number.isInteger(Number(status)) && Number(status) >= 200 && Number(status) <= 299));
})) throw new Error('Email provider explicitly rejected delivery.');
const source = $('Gate Durable Leave Reservation').all();
if (source.length !== 1 || sent.length !== 1) throw new Error('MAM email delivery count mismatch.');
const grant = receiptSingle(source);
if (grant.outcome !== 'send' || !receiptToken(grant.deliveryToken)
  || typeof grant.eventKey !== 'string' || !/^(leave:|payroll-balance:|payroll-comparison:)/.test(grant.eventKey)) receiptFail();
const result = receiptSingle(sent);
const identifier = (value) => typeof value === 'string' && value.length > 0 && value.length <= 1024
  && !/[\u0000-\u001f\u007f]/.test(value) ? value : null;
const status = result.statusCode ?? result.status;
const providerReceipt = receiptProvider({
  accepted: true,
  messageId: identifier(result.id) ?? identifier(result.messageId),
  requestId: identifier(result.headers?.['request-id']),
  statusCode: status === undefined || status === null ? null : Number(status),
});
// The next native HTTP node receives ONLY the minimal receipt contract.
return [{ json: { action: 'confirm', eventKey: grant.eventKey, deliveryToken: grant.deliveryToken,
  recipient: receiptRecipients(grant.deliveryRecipient), cc: receiptRecipients(grant.ccRecipients, false), providerReceipt } }];
