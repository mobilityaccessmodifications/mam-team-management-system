const source = $('Validate and Reserve Leave Notification').all();
const sent = $input.all();
if (sent.some(item=>{const r=item.json||{};const status=Number(r.statusCode||r.status);return r.error||r.success===false||r.ok===false||(Number.isFinite(status)&&status>=400);})) throw new Error('Email provider explicitly rejected delivery.');

const eventKey = String(source[0]?.json?.eventKey ?? '');
if (source.length !== 1 || sent.length !== 1 || !(eventKey.startsWith('leave:') || eventKey.startsWith('payroll-balance:') || eventKey.startsWith('payroll-comparison:'))) throw new Error('MAM email delivery count mismatch.');
const state = $getWorkflowStaticData('global');
const reservation = state.leaveNotificationEvents?.[eventKey];
const normalize = (value) => [...new Set(String(value ?? '').split(',').map(email => email.trim().toLowerCase()).filter(Boolean))].join(',');
const actualRecipient = normalize(source[0].json.deliveryRecipient);
const actualCcRecipients = normalize(source[0].json.ccRecipients);
if (source[0].json.outcome !== 'send' || !actualRecipient || reservation?.state !== 'processing'
  || reservation.deliveryRecipient !== actualRecipient || reservation.ccRecipients !== actualCcRecipients) {
  throw new Error('MAM email delivery reservation mismatch; reconciliation required.');
}
// Keep only provider acknowledgement metadata, never the raw response/mail/HR
// payload. Some sender nodes expose no message ID; null is deliberately honest.
const result = sent[0].json ?? {};
const identifier = (value) => typeof value === 'string' && value.length > 0 && value.length <= 1024
  && !/[\u0000-\u001f\u007f]/.test(value) ? value : null;
const status = Number(result.statusCode ?? result.status);
const providerReceipt = {
  accepted: true,
  messageId: identifier(result.id) ?? identifier(result.messageId),
  requestId: identifier(result.headers?.['request-id']),
  statusCode: Number.isInteger(status) && status >= 100 && status <= 599 ? status : null,
};
state.leaveNotificationEvents[eventKey] = { state: 'sent', at: Date.now(),
  actualRecipient, actualCcRecipients, providerReceipt };
return [{ json: {
  ok: true, sent: true, alreadySent: false,
  actualRecipient, actualCcRecipients, providerReceipt, eventKey,
} }];
