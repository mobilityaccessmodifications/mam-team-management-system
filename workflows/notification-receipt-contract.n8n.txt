// Prepended by deploy-notification-review.py to the durable receipt Code nodes.
// Errors deliberately contain no recipient, token, payload, or provider dump.
const receiptFail = () => { throw new Error('MAM durable receipt contract mismatch; reconciliation required.'); };
const receiptObject = value => value !== null && typeof value === 'object' && !Array.isArray(value);
const receiptKeys = (value, required, optional = []) => receiptObject(value)
  && required.every(key => Object.prototype.hasOwnProperty.call(value, key))
  && Object.keys(value).every(key => required.includes(key) || optional.includes(key));
const receiptSingle = items => {
  if (items.length !== 1 || !receiptObject(items[0]?.json)) receiptFail();
  return items[0].json;
};
const receiptToken = value => typeof value === 'string'
  && /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/i.test(value);
const receiptRecipients = (value, required = true) => {
  if (typeof value !== 'string' || /[\u0000-\u0020\u007f;]/.test(value)) receiptFail();
  if (!value && !required) return '';
  const list = value.split(',');
  if (list.length > 32 || list.some(email => email.length > 254 || !/^[^\s@,;]+@[^\s@,;]+\.[^\s@,;]+$/.test(email))) receiptFail();
  return [...new Set(list.map(email => email.toLowerCase()))].sort().join(',');
};
const receiptProvider = value => {
  const identifier = id => id === null || (typeof id === 'string' && id.length > 0
    && id.length <= 1024 && !/[\u0000-\u001f\u007f]/.test(id));
  if (!receiptKeys(value, ['accepted', 'messageId', 'requestId', 'statusCode']) || value.accepted !== true
    || !identifier(value.messageId) || !identifier(value.requestId)
    || !(value.statusCode === null || (Number.isInteger(value.statusCode) && value.statusCode >= 200 && value.statusCode <= 299))) receiptFail();
  return { accepted: true, messageId: value.messageId, requestId: value.requestId, statusCode: value.statusCode };
};
const receiptEvidence = value => {
  const actualRecipient = receiptRecipients(value.actualRecipient);
  const actualCcRecipients = receiptRecipients(value.actualCcRecipients, false);
  if (actualRecipient !== value.actualRecipient || actualCcRecipients !== value.actualCcRecipients) receiptFail();
  return { actualRecipient, actualCcRecipients, providerReceipt: receiptProvider(value.providerReceipt) };
};
